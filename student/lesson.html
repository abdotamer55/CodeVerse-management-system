تطبيق أمثلة Memoization وTabulation مع تمارين قصيرة داخل المختبر.

function fib(n, memo = {}) {
  if (n in memo) return memo[n];
  if (n <= 1) return n;
  return memo[n] = fib(n - 1, memo) + fib(n - 2, memo);
}
